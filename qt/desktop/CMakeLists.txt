# qt/desktop/CMakeLists.txt

project(ImaginatorDesktop LANGUAGES CXX)

message(STATUS "Configuring Imaginator Desktop UI...")

set(IMAGINATOR_DESKTOP_TARGET imaginator_desktop)

qt_add_executable(${IMAGINATOR_DESKTOP_TARGET})

set_target_properties(${IMAGINATOR_DESKTOP_TARGET} PROPERTIES
    AUTOMOC TRUE
    AUTORCC TRUE
    AUTOUIC TRUE
)

qt_add_resources(${IMAGINATOR_DESKTOP_TARGET} "config"
    PREFIX /
    FILES
       qtquickcontrols2.conf
)

set(QOBJECT_HEADERS 
)

set(CORE_SOURCES
)

set(CORE_HEADERS
)

set (QML_FILE
    # main
)

set (RESSOURCE_FILE
)

qt_add_qml_module(${IMAGINATOR_DESKTOP_TARGET}
    URI Imaginator.desktop
    VERSION 1.0
    QML_FILES
       ${QML_FILE}
       ${RESSOURCE_FILE}
)

target_sources(${IMAGINATOR_DESKTOP_TARGET}
    PRIVATE
       ${CORE_SOURCES}
       ${QOBJECT_HEADERS}

    PUBLIC
        FILE_SET HEADERS
        BASE_DIRS include
            ${CORE_HEADERS}
)

# Autogenerated bundle properties for macOS and iOS.
# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
set_target_properties(${IMAGINATOR_DESKTOP_TARGET} PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER org.imaginator.desktop
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR} #.${PROJECT_VERSION_MINOR} is not working
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

target_link_libraries(${IMAGINATOR_DESKTOP_TARGET}
    PRIVATE
        imaginator_qt_core_lib
)

include(GNUInstallDirs)
install(TARGETS imaginator_desktop
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

message(STATUS "Imaginator Desktop UI configured successfully.")
message(STATUS "   - Target: ${IMAGINATOR_DESKTOP_TARGET}")
message(STATUS "   - QML URI: Imaginator.desktop")
message(STATUS "")
